8. 小张量上 GPU 到底快不快

跑下面这段，把实际数字填进表里：

import torch, time

for n in [10, 100, 1000, 4000]:
    a = torch.randn(n, n)
    b = torch.randn(n, n)

    t = time.perf_counter()
    for _ in range(100):
        a @ b
    cpu = time.perf_counter() - t

    a, b = a.cuda(), b.cuda()
    torch.cuda.synchronize()
    t = time.perf_counter()
    for _ in range(100):
        a @ b
    torch.cuda.synchronize()
    gpu = time.perf_counter() - t

    print(f"n={n:<5} cpu={cpu:.4f}s  gpu={gpu:.4f}s  gpu/cpu={gpu/cpu:.2f}")
