五层卷积

import torch
import torch.nn as nn

class AlexNet(nn.Module):
    def __init__(self, num_classes=1000, in_channels=3):
        super().__init__()

        self.features = nn.Sequential(
            nn.Conv2d(in_channels=in_channels, out_channels=64,
                      kernel_size=11, stride=4, padding=2),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=3, stride=2),

            nn.Conv2d(in_channels=64, out_channels=192,
                      kernel_size=5, padding=2),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=3, stride=2),

            nn.Conv2d(in_channels=192, out_channels=384,
                      kernel_size=3, padding=1),
            nn.ReLU(inplace=True),

            nn.Conv2d(in_channels=384, out_channels=256,
                      kernel_size=3, padding=1),
            nn.ReLU(inplace=True),

            nn.Conv2d(in_channels=256, out_channels=256,
                      kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=3, stride=2),
        )

        self.classifier = nn.Sequential(
            nn.Dropout(p=0.5),
            nn.Linear(in_features=256 * 6 * 6, out_features=4096),
            nn.ReLU(inplace=True),
            nn.Dropout(p=0.5),
            nn.Linear(in_features=4096, out_features=4096),
            nn.ReLU(inplace=True),
            nn.Linear(in_features=4096, out_features=num_classes),
        )

    def forward(self, x):
        features = self.features(x)
        features = torch.flatten(features, start_dim=1)
        logits = self.classifier(features)
        return logits

小图版：用 MNIST 跑

ImageNet 下不了也跑不动。用 MNIST 跑一个同构的小版， 目的只是验证代码和流程。

改三处：

class AlexNetSmall(nn.Module):
    def __init__(self, num_classes=10):
        super().__init__()

        self.features = nn.Sequential(
            nn.Conv2d(in_channels=1, out_channels=64,      # ① 输入改 1 通道
                      kernel_size=11, stride=4, padding=2),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=3, stride=2),

            nn.Conv2d(64, 192, kernel_size=5, padding=2),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=3, stride=2),

            nn.Conv2d(192, 384, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(384, 256, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(256, 256, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
        )

        self.classifier = nn.Sequential(
            nn.AdaptiveAvgPool2d((1, 1)),                  # ② 不管多大都压成 1×1
            nn.Flatten(),
            nn.Dropout(p=0.4),
            nn.Linear(256, 1024),                          # ③ 缩小
            nn.ReLU(inplace=True),
            nn.Dropout(p=0.4),
            nn.Linear(1024, 1024),
            nn.ReLU(inplace=True),
            nn.Linear(1024, num_classes),
        )

    def forward(self, x):
        features = self.features(x)
        logits = self.classifier(features)
        return logits
AdaptiveAvgPool 是这里的关键零件

AdaptiveAvgPool2d((1, 1)) 的意思是不管进来多大，都压成 1×1。

32×32 的图经过五层卷积是 256@1×1，224×224 是 256@6×6—— 两种输入同一份代码都能跑。

这就省掉了「手算 Linear 该写多少」这个坑。

训练

device = "cuda" if torch.cuda.is_available() else "cpu"
model = AlexNetSmall().to(device)

print(sum(p.numel() for p in model.parameters()))    # 3,777,226

loss_fn = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)

for epoch in range(1, 7):
    model.train()
    for images, labels in train_loader:
        images = images.to(device)
        labels = labels.to(device)

        logits = model(images)
        loss = loss_fn(logits, labels)

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

    model.eval()
    val_loss, seen, correct = 0.0, 0, 0
    with torch.no_grad():
        for images, labels in val_loader:
            images = images.to(device)
            labels = labels.to(device)

            logits = model(images)
            val_loss += loss_fn(logits, labels).item() * labels.size(0)
            seen += labels.size(0)
            correct += (logits.argmax(dim=1) == labels).sum().item()

    print(
        f"{epoch}   val_loss {val_loss/seen:.4f}   "
        f"val_acc {100*correct/seen:.2f}%"
    )

三个观察

① 第 6 轮开始过拟合

轮val_lossval_acc
50.089397.90%
60.090297.63%

损失回升、准确率下降。正确做法是早停在第 5 轮。

② 它不如 LeNet

网络参数测试
LeNet61,70698.78%
AlexNet（小图版）3,777,22697.82%
61 倍参数，换来更差的准确率

这不是 bug，是结构要匹配数据规模。

MNIST 只有 28×28 灰度图，LeNet 已经够用。 AlexNet 的 11×11 大核、stride=4，是为 224×224 的复杂图设计的， 用在这里一上来就砍掉 3/4 的像素。

大模型不是万能药。

③ 为什么 AlexNet 用 ReLU

上一章实测过：同样 3 轮，sigmoid 版 94.42%，ReLU 版 98.31%。

AlexNet 的理由有两条：

理由说明
算得快max(0, x) 不用求幂，sigmoid 要算 e^x
梯度不衰减sigmoid 在两端梯度接近 0，ReLU 在正区间恒为 1

第二条是决定性的。深层网络里，sigmoid 的梯度连乘几十次就没了。
