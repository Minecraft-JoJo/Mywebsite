import torch.nn as nn

def count(*layers):
    return sum(p.numel() for l in layers for p in l.parameters())

def receptive_field(layers):
    rf, jump = 1, 1
    for kernel, stride in layers:
        rf += (kernel - 1) * jump
        jump *= stride
    return rf

C = 128

print("一个 11×11      ",
      count(nn.Conv2d(C, C, kernel_size=11, padding=5)))

print("两个 5×5 + 3×3  ",
      count(nn.Conv2d(C, C, 5, padding=2),
            nn.Conv2d(C, C, 5, padding=2),
            nn.Conv2d(C, C, 3, padding=1)))

print("五个 3×3        ",
      count(*[nn.Conv2d(C, C, 3, padding=1) for _ in range(5)]))

print()
print("感受野：",
      receptive_field([(11, 1)]),
      receptive_field([(5, 1), (5, 1), (3, 1)]),
      receptive_field([(3, 1)] * 5))

import torch.nn as nn

def conv_block(cin, cout):
    return nn.Sequential(
        nn.Conv2d(cin, cout, kernel_size=3, padding=1),
        nn.ReLU(inplace=True),
    )

CONFIG = [64, 64, "M", 128, 128, "M", 256, 256, 256, "M",
          512, 512, 512, "M", 512, 512, 512, "M"]

layers, ch = [], 3
for item in CONFIG:
    if item == "M":
        layers.append(nn.MaxPool2d(2, 2))
    else:
        layers.append(conv_block(ch, item))
        ch = item

features = nn.Sequential(*layers)
print(sum(p.numel() for p in features.parameters()))

四、辨真假

#说法对 / 错 + 理由
1VGG 的全称是「Very Good Grammar」
2三个 3×3 的感受野等于一个 7×7，参数更少
3VGG 的卷积层会改变特征图尺寸

5AlexNet 用 11×11 是因为大核比小核更强

参考答案

一、感受野与参数
方案感受野参数（C=128）ReLU
一个 11×11111,982,5921
两个 5×5 + 一个 3×311967,0403
五个 3×311737,9205

感受野都是 11，参数从 198 万降到 73.8 万，省 63%。

手算核对第一条：128 × 128 × 11 × 11 = 1,982,464， 加偏置 128 得 1,982,592。

乘 121 而不是 11——每个核是 11×11 的二维窗口。

「想一想」的答案：

参数省 63% 只是次要的。真正的原因是非线性次数： 五个 3×3 有 5 次 ReLU，一个 11×11 只有 1 次。

中间夹的非线性越多，网络能表达的决策边界越复杂。 这比省参数重要得多。

卷积 1471 万，全连接 1.24 亿——全连接还是占了 89%。

注意第 5 段：3 层 512→512 是 3 × (512×512×9 + 512) = 7,079,424， 占了卷积部分的一半。通道数的平方关系在这里体现得最明显。

三、读配置
问题答案
几层卷积5 层（每段一层）
几次池化5 次
224 最后多大224 ÷ 2⁵ = 7
和 VGG11 的区别每段只有 1 层卷积，通道翻倍更快但深度浅得多

这个配置其实更接近 AlexNet 的思路——每段只放一层。

所以 VGG 的贡献不是「3×3」，而是「每段放多层」。

四、辨真假
#答案理由
1错VGG 是牛津大学 Visual Geometry Group 的缩写，是实验室名字
2对感受野都是 7，参数 27C² vs 49C²
3错VGG 的卷积全部 padding=1，尺寸不变； 改变尺寸的是池化
4错层数从 8 到 16 翻倍，参数从 1.33 亿到 1.44 亿，只涨 8%
5错大核是为了快速缩小尺寸（11×11 步长 4，224 直接到 55）， 不是更强。VGG 证明小核堆叠更好
