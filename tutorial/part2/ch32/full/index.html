class Animals(Dataset):

        self.files = files
        self.labels = labels
        self.transform = transform

    def __len__(self):
        return len(self.files)

    def __getitem__(self, index):
        with Image.open(self.files[index]) as img:
            image = self.transform(img.convert("RGB"))

标准化必须用 ImageNet 的那一套

这里是预训练模型，不是从零训。

预训练时用的均值和方差是 ImageNet 的统计量， 微调时必须用同一套，否则激活值分布对不上。

尺寸也必须是 224×224——ResNet 就是在这个尺寸上学的。

mean = (0.485, 0.456, 0.406)

模型：冻结 + 换头

import torch
import torch.nn as nn
import torchvision.models as models

def build_model(freeze_backbone=True, num_classes=2):
    net = models.resnet18(weights=models.ResNet18_Weights.DEFAULT)

    if freeze_backbone:
        for param in net.parameters():
            param.requires_grad = False

    net.fc = nn.Linear(in_features=512, out_features=num_classes)
    return net

device = "cuda" if torch.cuda.is_available() else "cpu"
model = build_model(freeze_backbone=True).to(device)

def count_trainable(m):
    return sum(p.numel() for p in m.parameters() if p.requires_grad)

print(count_trainable(model))      # 1,026

训练

loss_fn = nn.CrossEntropyLoss()

trainable = [p for p in model.parameters() if p.requires_grad]
optimizer = torch.optim.Adam(trainable, lr=1e-3)

for epoch in range(1, 7):
    model.train()
    train_loss, seen, train_correct = 0.0, 0, 0
    for images, targets in train_loader:
        images = images.to(device)
        targets = targets.to(device)

        logits = model(images)
        loss = loss_fn(logits, targets)

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

        train_loss += loss.item() * targets.size(0)
        seen += targets.size(0)
        train_correct += (logits.argmax(dim=1) == targets).sum().item()

    model.eval()
    val_correct, val_seen = 0, 0
    with torch.no_grad():
        for images, targets in val_loader:
            images = images.to(device)
            targets = targets.to(device)
            val_correct += (model(images).argmax(dim=1) == targets).sum().item()
            val_seen += targets.size(0)

    print(
        f"{epoch}   loss {train_loss/seen:.4f}   "
        f"train {100*train_correct/seen:.2f}%   "
        f"val {100*val_correct/val_seen:.2f}%"
    )

结果对比

方案可训练参数最好验证测试集
冻结，只训 fc1,026100.00%100.00%

为什么训得少反而更准

因为数据只有 700 张。

11 亿个参数去学 700 张图，一定会把训练集背下来—— 训练准确率 98%，验证准确率在 87~99% 之间乱跳。

1026 个参数没能力背，只能学「猫和狗最本质的区别」， 反而泛化得更好。

全训版验证曲线为什么抖

87.33% → 99.33% → 97.33% → 87.33% → 91.33% → 96.67%。

两个原因：

原因说明
验证集只有 150 张错 2 张就是 1.3%
BatchNorm 统计量在变所有层都在训，BN 的滑动平均每轮都在更新

冻结版就没这个问题——所有 BN 层都冻住了，统计量固定， 所以稳定在 100%。

整段代码

合并起来，能直接跑：

import random
from pathlib import Path

import torch
import torch.nn as nn
import torchvision.models as models
from PIL import Image
from torch.utils.data import Dataset, DataLoader
from torchvision import transforms


# ── 1. 数据集 ──────────────────────────────
class Animals(Dataset):
    def __init__(self, files, labels, transform):
        self.files = files
        self.labels = labels
        self.transform = transform

    def __len__(self):
        return len(self.files)

    def __getitem__(self, index):
        with Image.open(self.files[index]) as img:
            image = self.transform(img.convert("RGB"))
        return image, self.labels[index]


source = Path("D:/main/archive/animals")
files, labels = [], []
for index, name in enumerate(["cat", "dog"]):
    for path in sorted((source / name).iterdir()):
        if path.suffix.lower() in (".png", ".jpg", ".jpeg"):
            files.append(path)
            labels.append(index)

random.seed(0)
order = list(range(len(files)))
random.shuffle(order)
n = len(order)
n_train = int(n * 0.70)
n_val = int(n * 0.15)

mean = (0.485, 0.456, 0.406)
std = (0.229, 0.224, 0.225)
train_tf = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.RandomHorizontalFlip(),
    transforms.ToTensor(),
    transforms.Normalize(mean, std),
])
eval_tf = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.ToTensor(),
    transforms.Normalize(mean, std),
])

def make_set(idx, tf):
    return Animals([files[i] for i in idx], [labels[i] for i in idx], tf)

train_set = make_set(order[:n_train], train_tf)
val_set = make_set(order[n_train:n_train + n_val], eval_tf)
test_set = make_set(order[n_train + n_val:], eval_tf)

train_loader = DataLoader(train_set, batch_size=16, shuffle=True)
val_loader = DataLoader(val_set, batch_size=32)
test_loader = DataLoader(test_set, batch_size=32)


# ── 2. 模型 ────────────────────────────────
def build_model(freeze_backbone=True, num_classes=2):
    net = models.resnet18(weights=models.ResNet18_Weights.DEFAULT)
    if freeze_backbone:
        for param in net.parameters():
            param.requires_grad = False
    net.fc = nn.Linear(in_features=512, out_features=num_classes)
    return net


device = "cuda" if torch.cuda.is_available() else "cpu"
model = build_model(freeze_backbone=True).to(device)


# ── 3. 训练 ────────────────────────────────
loss_fn = nn.CrossEntropyLoss()
trainable = [p for p in model.parameters() if p.requires_grad]
optimizer = torch.optim.Adam(trainable, lr=1e-3)

for epoch in range(1, 7):
    model.train()
    for images, targets in train_loader:
        images = images.to(device)
        targets = targets.to(device)

        logits = model(images)
        loss = loss_fn(logits, targets)

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

    model.eval()
    correct, total = 0, 0
    with torch.no_grad():
        for images, targets in val_loader:
            images = images.to(device)
            targets = targets.to(device)
            correct += (model(images).argmax(dim=1) == targets).sum().item()
            total += targets.size(0)

    print(f"{epoch}   val {100*correct/total:.2f}%")


# ── 4. 测试 ────────────────────────────────
model.eval()
correct, total = 0, 0
with torch.no_grad():
    for images, targets in test_loader:
        images = images.to(device)
        targets = targets.to(device)
        correct += (model(images).argmax(dim=1) == targets).sum().item()
        total += targets.size(0)

print(f"测试集 {100*correct/total:.2f}%")
